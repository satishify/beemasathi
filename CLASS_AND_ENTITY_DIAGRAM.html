Class diagram

classDiagram
    direction TB
    class Insurer {
        +str name
        +set hosts
        +list hubs
        +list sitemaps
        +set lines
        +str note
    }
    class Fetcher {
        +Session session
        +bool respect_robots
        +robots_allows(url)
        +get(url)
    }
    class PolicyVectorStore {
        +PersistentClient client
        +Collection collection
        +add_chunks(chunks)
        +replace_source_chunks(chunks)
        +search(query, insurer, policy_name, chunk_type)
        +get_catalog()
        +get_stats()
    }
    class PolicyRetriever {
        +PolicyVectorStore store
        +retrieve(query, insurer, policy_name, n_results)
        -_rerank(results, query)
        -_infer_line(query)
        -_infer_chunk_type(query)
    }
    class BimaSaathiGenerator {
        +str model
        +generate(chunks, question, catalog)
        -_select_chunks(chunks, question)
        -_safety_check(text)
        -_call_ollama(prompt, messages)
        -_catalog_fact_text(grouped, regulations)
    }
    class ClaimPredictor {
        +dict rules
        +predict(...) dict
        -_get_documents(procedure)
        -_build_summary(score, failed)
    }
    class QueryRequest {
        +str question
        +str insurer
        +str policy_name
    }
    class QueryResponse {
        +str answer
        +bool safe
        +list sources
    }
    class ClaimPredictRequest {
        +str insurer
        +str policy_name
        +bool has_ped
        +float room_rent_per_day
        +float sum_insured
    }
    class StreamlitApp {
        +retriever
        +generator
        +predictor
        +session_state messages
    }
    class FastAPIApp {
        +POST /ask
        +POST /predict
        +GET /health
    }
    class SyncKnowledge {
        +discover_files()
        +infer_metadata(path)
        +ingest_pdf(path, store)
        +ingest_text(path, store)
    }
    class HybridScraper {
        +run(insurer_keys, lines)
        +discover_documents(fetcher, key)
        +download_document(fetcher, doc)
    }
    class EvalJudge {
        +run_eval(case_ids)
        +_rule_score(case, result)
        +_call_judge(generator, case, result)
    }

    F --> G[PolicyRetriever]
    G --> H[BimaSaathiGenerator]
    H --> I[Streamlit UI]
    H --> J[FastAPI /ask]
    H --> K[eval/judge.py]
    L[ClaimPredictor] --> I
    L --> J
